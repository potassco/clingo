139class TermBaseMap {
140 public:
142 using Map = Util::ordered_map<SharedSymbol, prg_id_t>;
143
152 template <class F> auto add(Symbol sym, F &&fun) -> prg_id_t {
153 auto [it, ins] = map_.emplace(sym, 0);
154 if (ins) {
155 it.value() = std::invoke(std::forward<F>(fun));
156 }
157 return it.value();
158 }
159
166 void add(Symbol sym, prg_id_t id) {
167 if (!map_.emplace(sym, id).second) {
168 throw std::runtime_error("collision of term ids");
169 }
170 }
171
176 [[nodiscard]] auto term_id(size_t i) const -> prg_id_t {
177 if (i < size()) {
178 return map_.nth(i).value();
179 }
180 throw std::range_error{"index out of range"};
181 }
182
187 [[nodiscard]] auto symbol(size_t i) const -> Symbol {
188 if (i < size()) {
189 return *map_.nth(i).key();
190 }
191 throw std::range_error{"index out of range"};
192 }
193
201 [[nodiscard]] auto index(Symbol sym) const -> size_t { return map_.find(sym) - map_.begin(); }
202
206 [[nodiscard]] auto size() const -> size_t { return map_.size(); }
207
212 [[nodiscard]] auto begin() const -> Map::const_iterator { return map_.cbegin(); }
213
218 [[nodiscard]] auto end() const -> Map::const_iterator { return map_.cend(); }
219
220 private:
221 Util::ordered_map<SharedSymbol, prg_id_t> map_;
222};
