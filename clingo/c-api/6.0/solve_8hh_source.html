545 [](clingo_model_t *model, void *data, bool *goon) -> bool {
546 CLINGO_TRY {
547 auto *hnd = static_cast<SolveEventHandler *>(data);
548 assert(hnd != nullptr);
549 auto mdl = Model{model};
550 *goon = hnd->model(mdl);
551 return true;
552 }
553 CLINGO_CATCH;
554 },
555 [](int64_t const *values, size_t size, void *data) -> bool {
556 CLINGO_TRY {
557 auto *hnd = static_cast<SolveEventHandler *>(data);
558 assert(hnd != nullptr);
559 hnd->unsat({values, size});
560 }
561 CLINGO_CATCH;
562 },
563 [](clingo_stats_t *stats, void *data) -> bool {
564 CLINGO_TRY {
565 auto *hnd = static_cast<SolveEventHandler *>(data);
566 assert(hnd != nullptr);
567 uint64_t root = Detail::call<clingo_stats_root>(stats);
568 uint64_t step = Detail::call<clingo_stats_map_add_subkey>(
569 stats, root, clingo_user_stats_step.data, clingo_user_stats_step.size, clingo_stats_type_map);
570 uint64_t accu = Detail::call<clingo_stats_map_add_subkey>(
571 stats, root, clingo_user_stats_accu.data, clingo_user_stats_accu.size, clingo_stats_type_map);
572 hnd->stats(Stats{stats, step}, Stats{stats, accu});
573 }
574 CLINGO_CATCH;
575 },
576 [](clingo_solve_result_bitset_t result, void *data) -> void {
577 auto *hnd = static_cast<SolveEventHandler *>(data);
578 assert(hnd != nullptr);
