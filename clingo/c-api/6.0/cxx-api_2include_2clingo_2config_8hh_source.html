218 template <class Handler> void add(std::string_view name, std::string_view description, Handler &&handler) const {
219 using HandlerType = std::remove_cvref_t<Handler>;
220
221 static constexpr auto has_get = requires { handler.get(); };
222 static constexpr auto has_array_get = requires { handler.get(std::optional<size_t>{}); };
223 static constexpr auto has_set = requires { std::declval<HandlerType>().set(std::string_view{}); };
224 static constexpr auto has_array_set =
225 requires { std::declval<HandlerType>().set(std::optional<size_t>{}, std::string_view{}); };
226 static constexpr auto is_array = requires { handler.size(); };
227
228 static_assert(!has_get || !has_array_get, "entry cannot have both get with and without index");
229 static_assert(!has_set || !has_array_set, "entry cannot have both set with and without index");
230
231 static constexpr auto c_entry = clingo_config_entry_t{
232 has_get || has_array_get ? +[](size_t const *index, void *data, clingo_string_t *value, bool *has_value) -> bool {
233 CLINGO_TRY {
234 auto *self = static_cast<HandlerType *>(data);
235 thread_local std::optional<std::string> str;
236 if constexpr (has_array_get) {
237 str = self->get(index != nullptr ? std::optional{*index} : std::nullopt);
238 }
239 if constexpr (has_get) {
240 if (index != nullptr) {
241 throw std::logic_error{"cannot get array value"};
242 }
243 str = self->get();
244 }
245 if (str.has_value() && value != nullptr) {
246 value->data = str->data();
247 value->size = str->size();
248 }
249 if (has_value != nullptr) {
250 *has_value = str.has_value();
251 }
252 }
253 CLINGO_CATCH;
254 } : nullptr,
255 has_set || has_array_set ? +[](size_t const *index, char const *value, size_t size, void *data) -> bool {
256 CLINGO_TRY {
257 auto *self = static_cast<HandlerType *>(data);
258 if constexpr (has_array_set) {
259 self->set(index != nullptr ? std::optional{*index} : std::nullopt,
260 std::string_view{value, size});
261 }
262 if constexpr (has_set) {
263 if (index != nullptr) {
264 throw std::logic_error{"cannot set array value"};
265 }
266 self->set(std::string_view{value, size});
267 }
268 }
269 CLINGO_CATCH;
270 } : nullptr,
271 is_array ? +[](void *data, size_t *size, bool *has_size) -> bool {
272 CLINGO_TRY {
273 auto *self = static_cast<HandlerType *>(data);
274 if constexpr (is_array) {
275 if (size != nullptr) {
276 *size = self->size();
277 }
278 }
279 if (has_size != nullptr) {
280 *has_size = is_array;
281 }
282 }
283 CLINGO_CATCH;
284 } : nullptr,
285 [](void *data) { std::unique_ptr<HandlerType>{static_cast<HandlerType *>(data)}; },
286 };
287
288 auto copy = std::make_unique<HandlerType>(std::forward<Handler>(handler));
289 Detail::handle_error(clingo_config_add(cfg_(), key_, name.data(), name.size(), description.data(),
290 description.size(), &c_entry, copy.release()));
291 }
