115 [[nodiscard]] auto at(size_t index) const -> std::pair<Symbol, std::variant<int, double, Symbol>> {
116 clingo_symbol_t symbol = 0;
117 clingo_theory_value_t value;
118 bool found = false;
119 Detail::handle_error(theory_->assignment_get_value(theory_->self, thread_id_, index, &symbol, &value, &found));
120 auto sym = Clingo::Symbol{symbol, false};
121 if (!found) {
122 throw std::out_of_range{"invalid index"};
123 }
124 switch (value.type) {
125 case clingo_theory_value_type_int: {
126 return {std::move(sym), value.int_number}; // NOLINT
127 break;
128 }
129 case clingo_theory_value_type_double: {
130 return {std::move(sym), value.double_number}; // NOLINT
131 break;
132 }
133 case clingo_theory_value_type_symbol: {
134 return {std::move(sym), Symbol{value.symbol, false}}; // NOLINT
135 break;
136 }
137 default: {
