86inline auto store_error() -> bool {
87 try {
88 throw;
89 } catch (std::out_of_range const &e) {
90 auto msg = std::string_view{e.what()};
91 return clingo_set_error(clingo_result_range, msg.data(), msg.size());
92 } catch (std::invalid_argument const &e) {
93 auto msg = std::string_view{e.what()};
94 return clingo_set_error(clingo_result_invalid, msg.data(), msg.size());
95 } catch (std::bad_alloc const &e) {
96 auto msg = std::string_view{e.what()};
97 return clingo_set_error(clingo_result_bad_alloc, msg.data(), msg.size());
98 } catch (std::logic_error const &e) {
99 auto msg = std::string_view{e.what()};
100 return clingo_set_error(clingo_result_logic, msg.data(), msg.size());
101 } catch (std::exception const &e) {
102 auto msg = std::string_view{e.what()};
103 return clingo_set_error(clingo_result_runtime, msg.data(), msg.size());
104 } catch (...) {
105 auto msg = std::string_view{"no message"};
106 return clingo_set_error(clingo_result_runtime, msg.data(), msg.size());
107 }
108}

111inline void raise_error() {
112 clingo_string_t str;
113 clingo_result_t rc = clingo_result_success;
114 clingo_get_error(&rc, &str);
115 switch (rc) {
116 case clingo_result_bad_alloc: {
117 throw std::bad_alloc{};
118 }
119 case clingo_result_logic: {
120 throw std::logic_error{std::string{str.data, str.size}};
121 }
122 case clingo_result_invalid: {
123 throw std::invalid_argument{std::string{str.data, str.size}};
124 }
125 case clingo_result_range: {
126 throw std::out_of_range{std::string{str.data, str.size}};
127 }
128 default: {
129 throw std::runtime_error{std::string{str.data, str.size}};
130 }
131 }
132}
133
137inline void handle_error(bool res) {
138 if (!res) {
139 raise_error();
140 }
141}

143// Similar to handle_error but also reraises if the code is success.
144inline void handle_error_no_code(bool res, int code) {
145 if (!res) {
146 raise_error();
147 }
148 if ((code & 65) == 65 || (code & 33) == 33) { // NOLINT
149 clingo_string_t str;
150 clingo_result_t rc = clingo_result_success;
151 clingo_get_error(&rc, &str);
152 if (rc != clingo_result_success) {
153 raise_error();
154 }
155 }

158template <typename> struct funptr_traits;
159
160template <typename R, typename... As> struct funptr_traits<R (*)(As...)> {
161 using return_type = R;
162 using args_tuple = std::tuple<As...>;
163 template <std::size_t N> using arg = std::tuple_element_t<N, args_tuple>;
164
165 static constexpr std::size_t arity = sizeof...(As);
166 using last = arg<arity - 1>;
167};
168
169template <auto F, class... As> auto call(As &&...args) {
170 using Traits = funptr_traits<decltype(F)>;
171 auto res = std::remove_pointer_t<typename Traits::last>{};
172 if constexpr (std::is_same_v<typename Traits::return_type, void>) {
173 F(std::forward<As>(args)..., &res);
174 } else {
175 handle_error(F(std::forward<As>(args)..., &res));
176 }
177 return res;
178}

220 }

270template <class T, class P> class intrusive_handle {
271 public:
272 intrusive_handle() = default;
273 explicit intrusive_handle(P *ptr, bool inc = true) : ptr_{ptr} {
274 if (inc) {
275 T::acquire(ptr);
276 }
277 }
278 intrusive_handle(intrusive_handle const &other) : intrusive_handle{other.ptr_} {}
279 intrusive_handle(intrusive_handle &&other) noexcept : ptr_{std::exchange(other.ptr_, nullptr)} {}
280 ~intrusive_handle() noexcept { T::release(ptr_); }
281 // NOLINTNEXTLINE
282 auto operator=(intrusive_handle const &other) -> intrusive_handle & {
283 T::acquire(other.ptr_);
284 T::release(ptr_);
285 ptr_ = other.ptr_;
286 return *this;
287 }
288 auto operator=(intrusive_handle &&other) noexcept -> intrusive_handle & {
289 T::release(ptr_);
290 ptr_ = std::exchange(other.ptr_, nullptr);
291 return *this;
292 }
293 [[nodiscard]] auto get() const noexcept -> P * { return ptr_; }
294 void reset(std::nullptr_t = nullptr) noexcept {
295 T::release(ptr_);
296 ptr_ = nullptr;
297 }
298 void reset(P *ptr, bool inc = true) {
299 T::release(ptr_);
300 ptr_ = ptr;
301 if (inc) {
302 T::acquire(ptr_);
303 }
304 }
305 friend auto operator==(intrusive_handle const &p, std::nullptr_t) noexcept -> bool { return p.ptr_ == nullptr; }
306 friend auto operator!=(intrusive_handle const &p, std::nullptr_t) noexcept -> bool { return p.ptr_ != nullptr; }
307 friend auto operator==(std::nullptr_t, intrusive_handle const &p) noexcept -> bool { return p.ptr_ == nullptr; }

309

339template <typename T> struct UserDataTraits {
340 using StorageTraits = UserDataStorageTraits<std::remove_cvref_t<T>>;
341 using PointerType = StorageTraits::PtrType;
342 using ValueType = std::remove_pointer_t<PointerType>;
343 using ReferenceType = std::add_lvalue_reference_t<PointerType>;
344
346 static constexpr auto has_data = !requires { typename StorageTraits::is_null; };
347
349 template <typename U> static auto create(U &&data) -> StorageTraits::HolderType {
350 if constexpr (requires { typename StorageTraits::ValueType; }) {
351 return std::make_unique<typename StorageTraits::ValueType>(std::forward<U>(data));
352 } else if constexpr (requires { typename StorageTraits::is_ref; }) {
353 return &data.get();
354 } else {
355 return data;
356 }
357 }
359 static auto get(StorageTraits::HolderType &holder) -> PointerType {
360 if constexpr (requires { typename StorageTraits::is_holder; }) {
361 return holder.get();
362 } else {
363 return holder;
364 }
365 }
367 static auto has_value(StorageTraits::HolderType &holder) -> bool { return get(holder) != nullptr; }
369 static auto release(StorageTraits::HolderType &holder) -> StorageTraits::PtrType {
370 if constexpr (requires { typename StorageTraits::is_holder; }) {
371 return holder.release();
372 } else {
373 return holder;
374 }
375 }
377 static auto cast(void *ptr) -> StorageTraits::PtrType { return static_cast<StorageTraits::PtrType>(ptr); }
379 static void free(void *ptr) {
380 if constexpr (requires { typename StorageTraits::is_holder; }) {
381 typename StorageTraits::HolderType{cast(ptr)};
382 }
383 }

414template <class Seq> class RandomAccessIterator {
415 public:
416 using iterator_category = std::random_access_iterator_tag;
417 using value_type = Seq::value_type;
418 using size_type = Seq::size_type;
419 using difference_type = Seq::difference_type;
420 using pointer = Seq::pointer;
421 using reference = Seq::reference;
422
423 // NOTE: Added to fulfill the sentinel_for concept; should not be used.
424 constexpr RandomAccessIterator() : view_{throw std::logic_error("invalid iterator")}, index_{0} {}
425 constexpr RandomAccessIterator(Seq container, size_t index) noexcept : view_{std::move(container)}, index_{index} {}
426 constexpr auto operator*() const -> reference { return view_.at(index_); }
427 constexpr auto operator->() const -> pointer { return view_.at(index_); }
428 constexpr auto operator++() -> RandomAccessIterator & {
429 ++index_;
430 return *this;
431 }
432 constexpr auto operator++(int) -> RandomAccessIterator {
433 auto tmp = *this;
434 ++index_;
435 return tmp;
436 }
437 constexpr auto operator--() -> RandomAccessIterator & {
438 --index_;
439 return *this;
440 }
441 constexpr auto operator--(int) -> RandomAccessIterator {
442 auto tmp = *this;
443 --index_;
444 return tmp;
445 }
446 constexpr auto operator-(const RandomAccessIterator &other) const -> difference_type {
447 return static_cast<difference_type>(index_) - static_cast<difference_type>(other.index_);
448 }
449 constexpr auto operator+(difference_type n) const -> RandomAccessIterator {
450 return RandomAccessIterator(view_, index_ + n);
451 }
452 friend constexpr auto operator+(difference_type n, RandomAccessIterator it) -> RandomAccessIterator {
453 return RandomAccessIterator(it.view_, it.index_ + n);
454 }
455 constexpr auto operator-(difference_type n) const -> RandomAccessIterator {
456 return RandomAccessIterator(view_, index_ - n);
457 }
458 constexpr auto operator+=(difference_type n) -> RandomAccessIterator & {
459 index_ += n;
460 return *this;
461 }
462 constexpr auto operator-=(difference_type n) -> RandomAccessIterator & {
463 index_ -= n;
464 return *this;
465 }
466 constexpr auto operator==(const RandomAccessIterator &other) const -> bool { return index_ == other.index_; }
467 constexpr auto operator<=>(const RandomAccessIterator &other) const { return index_ <=> other.index_; }
468 constexpr auto operator[](difference_type n) const -> reference { return view_.at(index_ + n); }
469
470 private:
471 Seq view_;
472 size_t index_;
473};
