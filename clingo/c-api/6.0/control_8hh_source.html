413 auto profile() -> std::vector<ProfileNode> {
414 struct Builder {
415 static auto internal(size_t depth, char const *key, size_t key_size, bool nested, void *data) -> bool {
416 CLINGO_TRY {
417 auto *self = static_cast<Builder *>(data);
418 assert(depth <= self->stack.size());
419 self->stack.resize(depth);
420 auto node = ProfileNodeInternal{std::string{key, key_size}, nested};
421 if (self->stack.empty()) {
422 self->roots.emplace_back(std::move(node));
423 self->stack.emplace_back(&self->roots.back());
424 } else {
425 auto &children = std::get<ProfileNodeInternal>(*self->stack.back()).children;
426 children.emplace_back(std::move(node));
427 self->stack.emplace_back(&children.back());
428 }
429 }
430 CLINGO_CATCH;
431 }
432 static auto leaf(size_t depth, clingo_profile_data_t *values, clingo_profile_type_t type, void *data)
433 -> bool {
434 CLINGO_TRY {
435 auto *self = static_cast<Builder *>(data);
436 assert(depth <= self->stack.size());
437 self->stack.resize(depth);
438 auto node = ProfileNodeLeaf{static_cast<ProfileType>(type), values->matches, values->instances,
439 values->time_instantiate, values->time_propagate};
440 if (self->stack.empty()) {
441 self->roots.emplace_back(node);
442 } else {
443 std::get<ProfileNodeInternal>(*self->stack.back()).children.emplace_back(node);
444 }
445 }
446 CLINGO_CATCH;
447 }
448
449 std::vector<ProfileNode *> stack;
450 std::vector<ProfileNode> roots;
451 } builder;
452
453 auto visitor = clingo_profile_visitor_t{&Builder::internal, &Builder::leaf};
454 clingo_control_profile(ctl_.get(), &visitor, &builder);
455 return std::move(builder.roots);
456 }

610 template <Detail::UserData<Propagator, false> P> auto register_propagator(P &&propagator) const -> decltype(auto) {
611 using UserData = Detail::UserDataTraits<P>;
612 auto user_data = UserData::create(std::forward<P>(propagator));
613 if (!UserData::has_value(user_data)) {
614 throw std::invalid_argument{"propagator cannot be null"};
615 }
616 auto &res = *UserData::get(user_data);
617
618 static constexpr auto decide = [] {
619 if constexpr (std::is_base_of_v<Heuristic, typename UserData::ValueType>) {
620 return [](clingo_assignment_t const *assignment, clingo_literal_t fallback, void *data,
621 clingo_literal_t *decision) -> bool {
622 CLINGO_TRY {
623 *decision = UserData::cast(data)->decide(Assignment{assignment}, fallback);
624 }
625 CLINGO_CATCH;
626 };
627 } else {
628 return nullptr;
629 }
630 };
631
632 static constexpr auto c_propagator = clingo_propagator_t{
633 [](clingo_assignment_t const *assignment, clingo_propagate_init_t *init, void *data) -> bool {
634 CLINGO_TRY {
635 UserData::cast(data)->init(Assignment{assignment}, PropagateInit{init});
636 }
637 CLINGO_CATCH;
638 },
639 [](clingo_assignment_t const *assignment, clingo_propagate_control_t *control, void *data) {
640 CLINGO_TRY {
641 UserData::cast(data)->attach(Assignment{assignment}, PropagateControl{control});
642 }
643 CLINGO_CATCH;
644 },
645 [](clingo_assignment_t const *assignment, clingo_propagate_control_t *control,
646 clingo_literal_t const *changes, size_t size, void *data) -> bool {
647 CLINGO_TRY {
648 UserData::cast(data)->propagate(Assignment{assignment}, PropagateControl{control},
649 SolverLiteralSpan{changes, size});
650 }
651 CLINGO_CATCH;
652 },
653 [](clingo_assignment_t const *assignment, clingo_literal_t const *changes, size_t size, void *data) {
654 try {
655 UserData::cast(data)->undo(Assignment{assignment}, SolverLiteralSpan{changes, size});
656 } catch (std::exception const &e) {
657 fprintf(stderr, "panic: %s\n", e.what());
658 std::terminate();
659 }
660 },
661 [](clingo_assignment_t const *assignment, clingo_propagate_control_t *control, void *data) -> bool {
662 CLINGO_TRY {
663 UserData::cast(data)->check(Assignment{assignment}, PropagateControl{control});
664 }
665 CLINGO_CATCH;
666 },
667 decide(),
668 [](void *data) { UserData::free(data); },
669 };
670
671 Detail::handle_error(
672 clingo_control_register_propagator(ctl_.get(), &c_propagator, UserData::release(user_data)));
673 return res;
674 }

688 template <Detail::UserData<SolveEventHandler> Handler = std::nullptr_t>
689 [[nodiscard]] auto solve_(Handler &&handler, ProgramLiteralSpan const &assumptions, SolveFlags flags) const
690 -> SolveHandle {
691 using UserData = Detail::UserDataTraits<Handler>;
692 auto user_data = UserData::create(std::forward<Handler>(handler));
693 clingo_solve_event_handler_t const *c_handler_ptr = nullptr;
694 if constexpr (UserData::has_data) {
695 if (UserData::has_value(user_data)) {
696 static constexpr auto c_solve_event_handler = clingo_solve_event_handler_t{
697 [](clingo_model_t *model, void *data, bool *goon) -> bool {
698 CLINGO_TRY {
699 auto *hnd = UserData::cast(data);
700 assert(hnd != nullptr);
701 auto mdl = Model{model};
702 *goon = hnd->model(mdl);
703 return true;
704 }
705 CLINGO_CATCH;
706 },
707 [](int64_t const *values, size_t size, void *data) -> bool {
708 CLINGO_TRY {
709 auto *hnd = UserData::cast(data);
710 assert(hnd != nullptr);
711 hnd->unsat({values, size});
712 }
713 CLINGO_CATCH;
714 },
715 [](clingo_stats_t *stats, void *data) -> bool {
716 CLINGO_TRY {
717 auto *hnd = UserData::cast(data);
718 assert(hnd != nullptr);
719 uint64_t root = Detail::call<clingo_stats_root>(stats);
720 uint64_t step = Detail::call<clingo_stats_map_add_subkey>(
721 stats, root, clingo_user_stats_step.data, clingo_user_stats_step.size,
722 clingo_stats_type_map);
723 uint64_t accu = Detail::call<clingo_stats_map_add_subkey>(
724 stats, root, clingo_user_stats_accu.data, clingo_user_stats_accu.size,
725 clingo_stats_type_map);
726 hnd->stats(Stats{stats, step}, Stats{stats, accu});
727 }
728 CLINGO_CATCH;
729 },
730 [](clingo_solve_result_bitset_t result, void *data) -> void {
731 auto *hnd = UserData::cast(data);
732 assert(hnd != nullptr);
733 hnd->finish(static_cast<SolveResult>(result));
734 },
735 [](void *data) { UserData::free(data); },
736 };
737 c_handler_ptr = &c_solve_event_handler;
738 }
739 }
740
741 clingo_solve_handle_t *res = nullptr;
742 Detail::handle_error(clingo_control_solve(ctl_.get(), static_cast<clingo_solve_mode_bitset_t>(flags),
743 assumptions.data(), assumptions.size(), c_handler_ptr,
744 UserData::release(user_data), &res));
745 return SolveHandle{res};
746 }

755 [[nodiscard]] auto c_parts_(std::optional<PartSpan> parts = std::nullopt) const
756 -> std::pair<std::optional<PartVector>, std::vector<clingo_part_t>> {
757 auto res = std::pair<std::optional<PartVector>, std::vector<clingo_part_t>>{};
758 auto &[default_parts, c_parts] = res;

761 }
762 if (!parts) {
763 static auto default_part = Part{"base", {}};
764 parts = PartSpan{&default_part, 1};
765 }
766 assert(parts);
767 c_parts.reserve(parts->size());
768 for (auto const &part : *parts) {
769 c_parts.emplace_back(part.name.data(), part.name.size(), c_cast(part.params.data()), part.params.size());
770 }
771 return res;
