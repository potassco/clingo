97auto transform_vec(std::optional<std::vector<T>> const &vec, F const &f)
98 -> Detail::transform_vec_result<T const &, F const &> {
99 return transform(vec, [&f](auto const &vec) {
100 typename Detail::transform_vec_result<T const &, F const &>::value_type ret;
101 ret.reserve(vec.size());
102 for (auto const &elem : vec) {
103 ret.emplace_back(std::invoke(f, elem));
104 }
105 return ret;
106 });

150template <class T, bool UseSpan = true> class ResultVec {
151 public:
153 using ValueType = T;
155 using Span = std::span<ValueType const>;
157 using Array = Util::immutable_array<ValueType>;
159 using Vector = std::vector<ValueType>;
161 using Source = std::conditional_t<UseSpan, Span, Array const &>;
163 using Iterator = std::conditional_t<UseSpan, typename Span::iterator, typename Array::const_iterator>;
165 using Result = std::conditional_t<UseSpan, Vector, Array>;
166
168 explicit ResultVec(Source source) : source_{source}, current_{source_.begin()} {}
169
171 [[nodiscard]] auto current() const -> ValueType const & { return *current_; }
172
174 void keep() {
175 if (result_) {
176 result_->emplace_back(*current_);
177 }
178 ++current_;
179 }
181 void keep_all() {
182 if (result_) {
183 result_->insert(result_->end(), current_, source_.end());
184 }
185 current_ = source_.end();
186 }
188 void remove() {
189 if (!result_) {
190 result_ = Util::copy_n(source_, std::distance(source_.begin(), current_));
191 }
192 ++current_;
193 }
195 template <class... Args> void replace(Args &&...args) {
196 if (!result_) {
197 result_ = Util::copy_n(source_, std::distance(source_.begin(), current_));
198 }
199 result_->emplace_back(std::forward<Args>(args)...);
200 ++current_;
201 }
203 void update(std::optional<ValueType> value) {
204 if (!value.has_value()) {
205 keep();
206 } else {
207 replace(*std::move(value));
208 }
209 }
211 template <class... Args> void append(Args &&...args) {
212 if (!result_) {
213 result_ = Util::copy_n(source_, std::distance(source_.begin(), current_));
214 }
215 result_->emplace_back(std::forward<Args>(args)...);
216 }
218 template <class It> void extend(It begin, It end) {
219 if (!result_) {
220 result_ = Util::copy_n(source_, std::distance(source_.begin(), current_));
221 }
222 result_->insert(result_->end(), begin, end);
223 }
227 [[nodiscard]] auto value() const & -> Span {
228 if (result_) {
229 return *result_;
230 }
231 return source_;
232 }
234 [[nodiscard]] auto value() && -> Result {
235 if (result_) {
236 return *std::move(result_);
237 }
238 if constexpr (UseSpan) {
239 return {source_.begin(), source_.end()};
240 } else {
241 return source_;
242 }
243 }
245 [[nodiscard]] auto has_value() const -> bool { return result_.has_value(); }
247 [[nodiscard]] auto as_optional() & -> std::optional<Vector> & { return result_; }
249 [[nodiscard]] auto as_optional() && -> std::optional<Vector> { return std::move(result_); }
250
254 [[nodiscard]] auto operator*() const & -> Span { return value(); }
256 [[nodiscard]] auto operator*() && -> Result { return std::move(*this).value(); }
258 explicit operator bool() const { return has_value(); }
260 [[nodiscard]] auto complete() const { return current_ == source_.end(); }
261
262 private:
263 Source source_;
264 std::optional<Vector> result_;
265 Iterator current_;
266};
