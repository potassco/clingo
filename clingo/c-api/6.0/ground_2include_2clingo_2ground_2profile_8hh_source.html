175template <typename T> class ProfileNodeExpression : public ProfileNodeInternal {
176 public:
178 ProfileNodeExpression(T expr, bool nested = false) : expr_{std::move(expr)}, nested_{nested} {}
179
180 private:
182 auto do_add_child(std::unique_ptr<ProfileNode> child) -> ProfileNode * override {
183 for (auto const &x : children_) {
184 if (x->equal(*child)) {
185 return x.get();
186 }
187 }
188 children_.emplace_back(std::move(child));
189 return children_.back().get();
190 }
191
193 void do_print(std::ostream &out, ProfileIndent indent, ProfileDetail detail, ProfileType type) const override {
194 out << indent << (nested_ ? "[" : "") << expr_ << (nested_ ? "]" : "") << "\n";
195 if (detail == ProfileDetail::detailed) {
196 for (auto const &child : children_) {
197 child->print(out, indent + 1, detail, type);
198 }
199 } else {
200 auto stats = ProfileStats{};
201 combine(stats, type, false);
202 stats.print(out, indent + 1);
203 }
204 }
205
207 [[nodiscard]] auto do_equal(ProfileNode const &node) const -> bool override {
208 auto const *other = dynamic_cast<ProfileNodeExpression const *>(&node);
209 if constexpr (requires { expr_.get(); }) {
210 return other != nullptr && nested_ == other->nested_ && expr_.get() == other->expr_.get();
211 } else {
212 return other != nullptr && nested_ == other->nested_ && expr_ == other->expr_;
213 }
214 }
215
217 [[nodiscard]] auto do_score(ProfileType type) const -> double override {
218 return !nested_ ? std::accumulate(children_.begin(), children_.end(), 0.0,
219 [=](double sum, auto const &child) { return sum + child->score(type); })
220 : 0;
221 }
222
224 void do_begin_step() override {
225 for (auto &child : children_) {
226 child->begin_step();
227 }
228 }
229
231 void do_end_step() override {
232 for (auto &child : children_) {
233 child->end_step();
234 }
235 }
236
238 void do_combine(ProfileStats &stats, ProfileType type, bool nested) const override {
239 for (auto const &child : children_) {
240 child->combine(stats, type, nested || nested_);
241 }
242 }
243
245 void do_accept(Visitor const &visit, size_t depth) const override {
246 std::ostringstream oss;
247 oss << expr_;
248 visit(std::make_pair(oss.view(), nested_), depth);
249 for (auto const &child : children_) {
250 child->accept(visit, depth + 1);
251 }
252 }
253
255 T expr_;
257 std::vector<std::unique_ptr<ProfileNode>> children_;
259 bool nested_;
260};
