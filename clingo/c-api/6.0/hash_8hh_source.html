196inline auto hash_combine(size_t seed, size_t h) -> size_t {
197 // NOLINTBEGIN(readability-magic-numbers)
198 if constexpr (sizeof(size_t) == sizeof(uint64_t)) {
199 seed *= 0x87c37b91114253d5;
200 seed = (seed >> 31) | (seed << 33);
201 seed *= 0x4cf5ad432745937f;
202 seed ^= h;
203 seed = (seed >> 27) | (seed << 37);
204 seed = (seed * 5) + 0x52dce729;
205 } else {
206 seed *= 0xcc9e2d51;
207 seed = (seed >> 17) | (seed << 15);
208 seed *= 0x1b873593;
209 seed ^= h;
210 seed = (seed >> 19) | (seed << 13);
211 seed = (seed * 5) + 0xe6546b64;
212 }
213 return seed;
214 // NOLINTEND(readability-magic-numbers)
215}

223inline auto hash_mix(size_t h) -> size_t {
224 // NOLINTBEGIN(readability-magic-numbers)
225 if constexpr (sizeof(size_t) == sizeof(uint64_t)) {
226 h ^= h >> 33;
227 h *= 0xff51afd7ed558ccd;
228 h ^= h >> 33;
229 h *= 0xc4ceb9fe1a85ec53;
230 h ^= h >> 33;
231 } else {
232 h ^= h >> 16;
233 h *= 0x85ebca6b;
234 h ^= h >> 13;
235 h *= 0xc2b2ae35;
236 h ^= h >> 16;
237 }
238 return h;
239 // NOLINTEND(readability-magic-numbers)
240}

246template <class T> auto value_hash_range(T const &x) -> size_t {
247 // NOLINTBEGIN(readability-magic-numbers)
248 // note: we unroll the loop for small sizes, this compiles very well and leads to noticable speedups
249 // length 8 should cover typical sizes for argument tuples etc. in clingo
250 auto small = []<std::size_t... n>(std::index_sequence<n...>, auto const &x) {
251 return Util::hash_combine(Util::value_hash(x[n])...);
252 };
253 switch (x.size()) {
254 case 0:
255 return small(std::make_index_sequence<0>(), x);
256 case 1:
257 return small(std::make_index_sequence<1>(), x);
258 case 2:
259 return small(std::make_index_sequence<2>(), x);
260 case 3:
261 return small(std::make_index_sequence<3>(), x);
262 case 4:
263 return small(std::make_index_sequence<4>(), x);
264 case 5:
265 return small(std::make_index_sequence<5>(), x);
266 case 6:
267 return small(std::make_index_sequence<6>(), x);
268 case 7:
269 return small(std::make_index_sequence<7>(), x);
270 case 8:
271 return small(std::make_index_sequence<8>(), x);
272 default:
273 return std::accumulate(
274 x.begin() + 1, x.end(), Util::value_hash(*x.begin()),
275 [](auto const &seed, auto const &x) { return Util::hash_combine(seed, Util::value_hash(x)); });
276 }
277 // NOLINTEND(readability-magic-numbers)
278}
