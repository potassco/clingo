cmake_minimum_required(VERSION 3.22.1)

project(CLINGO VERSION "1.0" LANGUAGES C CXX)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(macros)
include(CMakeDependentOption)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(PARSER_BUILD_WEB "compile to javascript with emscripten" OFF)
option(PARSER_PROFILE "configure profiling" OFF)
option(PARSER_BUILD_TESTS "enable tests" ON)
option(PARSER_BUILD_APP "build application" ON)
option(PARSER_BUILD_EXAMPLES "build examples" ON)
cmake_dependent_option(PARSER_BUILD_SHARED "build clingo as shared library" ON "NOT SKBUILD" OFF)
cmake_dependent_option(PARSER_USE_VERSION_SCRIPT "use a version script to hide C++ symbols" ON "PARSER_BUILD_SHARED OR SKBUILD" OFF)
cmake_dependent_option(PARSER_USE_VISIBILITY "enable visibility attributes" ON "SKBUILD OR PARSER_BUILD_SHARED" OFF)

find_package(RE2C "2.0" REQUIRED)

if (PARSER_USE_VISIBILITY)
    set(CMAKE_C_VISIBILITY_PRESET hidden)
    set(CMAKE_CXX_VISIBILITY_PRESET hidden)
    set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
endif()

if (PARSER_BUILD_WEB)
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -s DISABLE_EXCEPTION_CATCHING=0")
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -s ENVIRONMENT=worker,node -s ALLOW_MEMORY_GROWTH=1 -s INVOKE_RUN=1")
    set(CMAKE_C_FLAGS_RELEASE "-Os -DNDEBUG")
    set(CMAKE_CXX_FLAGS_RELEASE "-Os -DNDEBUG")
    set(DCMAKE_EXE_LINKER_FLAGS_RELEASE "")
endif()

if (PARSER_BUILD_TESTS)
    set(LIB_POTASSCO_SETUP_CATCH2 TRUE)
    enable_testing()
endif()

if (PARSER_PROFILE)
    find_package(Gperftools REQUIRED)
endif()

if (DEFINED SKBUILD)
    set(CLASP_BUILD_APP OFF)
    set(PARSER_BUILD_TESTS OFF)
    set(PARSER_BUILD_WEB OFF)
    set(PARSER_BUILD_APP OFF)
    set(PARSER_BUILD_EXAMPLES OFF)
    set(PYBIND11_NEWPYTHON ON)
    set(PYBIND11_FINDPYTHON ON)
    find_package(pybind11 CONFIG REQUIRED)
endif()

if (EMSCRIPTEN)
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fexceptions")
endif()

add_subdirectory(third_party)
add_subdirectory(lib libclingo)
if (PARSER_BUILD_EXAMPLES)
    add_subdirectory(examples/c-api)
endif()
if (PARSER_BUILD_APP)
    add_subdirectory(app)
endif()
if (PARSER_BUILD_WEB)
    add_subdirectory(web)
endif()
