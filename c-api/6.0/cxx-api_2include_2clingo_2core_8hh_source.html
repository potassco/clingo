82inline auto store_error() -> bool {
83 try {
84 throw;
85 } catch (std::out_of_range const &e) {
86 auto msg = std::string_view{e.what()};
87 return clingo_set_error(clingo_result_range, msg.data(), msg.size());
88 } catch (std::invalid_argument const &e) {
89 auto msg = std::string_view{e.what()};
90 return clingo_set_error(clingo_result_invalid, msg.data(), msg.size());
91 } catch (std::bad_alloc const &e) {
92 auto msg = std::string_view{e.what()};
93 return clingo_set_error(clingo_result_bad_alloc, msg.data(), msg.size());
94 } catch (std::logic_error const &e) {
95 auto msg = std::string_view{e.what()};
96 return clingo_set_error(clingo_result_logic, msg.data(), msg.size());
97 } catch (std::exception const &e) {
98 auto msg = std::string_view{e.what()};
99 return clingo_set_error(clingo_result_runtime, msg.data(), msg.size());
100 } catch (...) {
101 auto msg = std::string_view{"no message"};
102 return clingo_set_error(clingo_result_runtime, msg.data(), msg.size());
103 }
104}

107inline void raise_error() {
108 clingo_string_t str;
109 clingo_result_t rc = clingo_result_success;
110 clingo_get_error(&rc, &str);
111 switch (rc) {
112 case clingo_result_bad_alloc: {
113 throw std::bad_alloc{};
114 }
115 case clingo_result_logic: {
116 throw std::logic_error{std::string{str.data, str.size}};
117 }
118 case clingo_result_invalid: {
119 throw std::invalid_argument{std::string{str.data, str.size}};
120 }
121 case clingo_result_range: {
122 throw std::out_of_range{std::string{str.data, str.size}};
123 }
124 default: {
125 throw std::runtime_error{std::string{str.data, str.size}};
126 }
127 }
128}

133inline void handle_error(bool res) {
134 if (!res) {
135 raise_error();
136 }

139// Similar to handle_error but also reraises if the code is success.
140inline void handle_error_no_code(bool res, int code) {
141 if (!res) {
142 raise_error();
143 }
144 if ((code & 65) == 65 || (code & 33) == 33) { // NOLINT
145 clingo_string_t str;
146 clingo_result_t rc = clingo_result_success;
147 clingo_get_error(&rc, &str);
148 if (rc != clingo_result_success) {
149 raise_error();
150 }
151 }
152}

154template <typename> struct funptr_traits;
155
156template <typename R, typename... As> struct funptr_traits<R (*)(As...)> {
157 using return_type = R;
158 using args_tuple = std::tuple<As...>;
159 template <std::size_t N> using arg = std::tuple_element_t<N, args_tuple>;
160
161 static constexpr std::size_t arity = sizeof...(As);
162 using last = arg<arity - 1>;
163};
164
165template <auto F, class... As> auto call(As &&...args) {
166 using Traits = funptr_traits<decltype(F)>;
167 auto res = std::remove_pointer_t<typename Traits::last>{};
168 if constexpr (std::is_same_v<typename Traits::return_type, void>) {
169 F(std::forward<As>(args)..., &res);
170 } else {
171 handle_error(F(std::forward<As>(args)..., &res));
172 }
173 return res;
174}

209 if (this != &other) {
210 if (*this) {
211 T::free(ptr_);
212 }
213 ptr_ = std::exchange(other.ptr_, nullptr);
214 }
215 return *this;
216 }

266template <class T, class P> class intrusive_handle {
267 public:
268 intrusive_handle() = default;
269 explicit intrusive_handle(P *ptr, bool inc = true) : ptr_{ptr} {
270 if (inc) {
271 T::acquire(ptr);
272 }
273 }
274 intrusive_handle(intrusive_handle const &other) : intrusive_handle{other.ptr_} {}
275 intrusive_handle(intrusive_handle &&other) noexcept : ptr_{std::exchange(other.ptr_, nullptr)} {}
276 ~intrusive_handle() noexcept { T::release(ptr_); }
277 auto operator=(intrusive_handle const &other) -> intrusive_handle & {
278 T::acquire(other.ptr_);
279 T::release(ptr_);
280 ptr_ = other.ptr_;
281 return *this;
282 }
283 auto operator=(intrusive_handle &&other) noexcept -> intrusive_handle & {
284 T::release(ptr_);
285 ptr_ = std::exchange(other.ptr_, nullptr);
286 return *this;
287 }
288 [[nodiscard]] auto get() const noexcept -> P * { return ptr_; }
289 void reset(std::nullptr_t = nullptr) noexcept {
290 T::release(ptr_);
291 ptr_ = nullptr;
292 }
293 void reset(P *ptr, bool inc = true) {
294 T::release(ptr_);
295 ptr_ = ptr;
296 if (inc) {
297 T::acquire(ptr_);
298 }
299 }
300 friend auto operator==(intrusive_handle const &p, std::nullptr_t) noexcept -> bool { return p.ptr_ == nullptr; }
301 friend auto operator!=(intrusive_handle const &p, std::nullptr_t) noexcept -> bool { return p.ptr_ != nullptr; }
302 friend auto operator==(std::nullptr_t, intrusive_handle const &p) noexcept -> bool { return p.ptr_ == nullptr; }
303 friend auto operator!=(std::nullptr_t, intrusive_handle const &p) noexcept -> bool { return p.ptr_ != nullptr; }
304
305 private:
306 P *ptr_ = nullptr;
307};

334template <typename T> struct UserDataTraits {
335 using StorageTraits = UserDataStorageTraits<std::remove_cvref_t<T>>;
336 using PointerType = typename StorageTraits::PtrType;
337 using ValueType = std::remove_pointer_t<PointerType>;
338 using ReferenceType = std::add_lvalue_reference_t<PointerType>;
339
341 static constexpr auto has_data = !requires { typename StorageTraits::is_null; };
342
344 template <typename U> static auto create(U &&data) -> StorageTraits::HolderType {
345 if constexpr (requires { typename StorageTraits::ValueType; }) {
346 return std::make_unique<typename StorageTraits::ValueType>(std::forward<U>(data));
347 } else if constexpr (requires { typename StorageTraits::is_ref; }) {
348 return &data.get();
349 } else {
350 return data;
351 }
352 }
354 static auto get(StorageTraits::HolderType &holder) -> PointerType {
355 if constexpr (requires { typename StorageTraits::is_holder; }) {
356 return holder.get();
357 } else {
358 return holder;
359 }
360 }
362 static auto has_value(StorageTraits::HolderType &holder) -> bool { return get(holder) != nullptr; }
364 static auto release(StorageTraits::HolderType &holder) -> StorageTraits::PtrType {
365 if constexpr (requires { typename StorageTraits::is_holder; }) {
366 return holder.release();
367 } else {
368 return holder;
369 }
370 }
372 static auto cast(void *ptr) -> StorageTraits::PtrType { return static_cast<StorageTraits::PtrType>(ptr); }
374 static void free(void *ptr) {
375 if constexpr (requires { typename StorageTraits::is_holder; }) {
376 typename StorageTraits::HolderType{cast(ptr)};
377 }
378 }
379};

409template <class Seq> class RandomAccessIterator {
410 public:
411 using iterator_category = std::random_access_iterator_tag;
412 using value_type = typename Seq::value_type;
413 using size_type = typename Seq::size_type;
414 using difference_type = typename Seq::difference_type;
415 using pointer = typename Seq::pointer;
416 using reference = typename Seq::reference;
417
418 // NOTE: Added to fulfill the sentinel_for concept; should not be used.
419 constexpr RandomAccessIterator() : view_{throw std::logic_error("invalid iterator")}, index_{0} {}
420 constexpr RandomAccessIterator(Seq container, size_t index) noexcept : view_{std::move(container)}, index_{index} {}
421 constexpr auto operator*() const -> reference { return view_.at(index_); }
422 constexpr auto operator->() const -> pointer { return view_.at(index_); }
423 constexpr auto operator++() -> RandomAccessIterator & {
424 ++index_;
425 return *this;
426 }
427 constexpr auto operator++(int) -> RandomAccessIterator {
428 auto tmp = *this;
429 ++index_;
430 return tmp;
431 }
432 constexpr auto operator--() -> RandomAccessIterator & {
433 --index_;
434 return *this;
435 }
436 constexpr auto operator--(int) -> RandomAccessIterator {
437 auto tmp = *this;
438 --index_;
439 return tmp;
440 }
441 constexpr auto operator-(const RandomAccessIterator &other) const -> difference_type {
442 return static_cast<difference_type>(index_) - static_cast<difference_type>(other.index_);
443 }
444 constexpr auto operator+(difference_type n) const -> RandomAccessIterator {
445 return RandomAccessIterator(view_, index_ + n);
446 }
447 friend constexpr auto operator+(difference_type n, RandomAccessIterator it) -> RandomAccessIterator {
448 return RandomAccessIterator(it.view_, it.index_ + n);
449 }
450 constexpr auto operator-(difference_type n) const -> RandomAccessIterator {
451 return RandomAccessIterator(view_, index_ - n);
452 }
453 constexpr auto operator+=(difference_type n) -> RandomAccessIterator & {
454 index_ += n;
455 return *this;
456 }
457 constexpr auto operator-=(difference_type n) -> RandomAccessIterator & {
458 index_ -= n;
459 return *this;
460 }
461 constexpr auto operator==(const RandomAccessIterator &other) const -> bool { return index_ == other.index_; }
462 constexpr auto operator<=>(const RandomAccessIterator &other) const { return index_ <=> other.index_; }
463 constexpr auto operator[](difference_type n) const -> reference { return view_.at(index_ + n); }
464
465 private:
466 Seq view_;
467 size_t index_;
468};
