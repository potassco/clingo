100 [[nodiscard]] auto operator[](size_t index) const -> T {
101 assert(index < size() && last_ < values_.size());
102 auto l = last_ > 0 ? values_[last_ - 1].first : size_t{0};
103 auto r = values_[last_].first;
104 assert(l < r);
105 auto ib = index < l ? values_.begin() : values_.begin() + last_;
106 auto ie = index < r ? values_.begin() + last_ + 1 : values_.end();
107 auto it = std::upper_bound(ib, ie, index, [](auto const &a, auto const &b) { return a < b.first; });
108 last_ = static_cast<size_t>(std::distance(values_.begin(), it));
109 auto start = last_ > 0 ? values_[last_ - 1].first : size_t{0};
110 return it->second + static_cast<T>(index - start);
